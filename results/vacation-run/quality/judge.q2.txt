// ─── Token Bucket ────────────────────────────────────────────────────────────

/**
 * Async token-bucket rate limiter.
 *
 * Serialises acquire() calls internally to guarantee fairness: waiting
 * callers are served in FIFO order and each consumes exactly one token,
 * so the effective throughput never exceeds `maxRPS`.
 */
class TokenBucket {
  private tokens: number;
  private lastRefill: number;
  private readonly capacity: number;
  private readonly refillRatePerMs: number;

  // FIFO serialisation chain
  private tail: Promise<void> = Promise.resolve();

  constructor(
    private readonly maxRPS: number,
    capacity?: number,
  ) {
    this.capacity = capacity ?? maxRPS;
    this.tokens = this.capacity;
    this.lastRefill = performance.now();
    this.refillRatePerMs = maxRPS / 1000;
  }

  private refill(): void {
    const now = performance.now();
    const elapsed = now - this.lastRefill;
    if (elapsed <= 0) return;
    this.tokens = Math.min(this.capacity, this.tokens + elapsed * this.refillRatePerMs);
    this.lastRefill = now;
  }

  /**
   * Acquires one token, sleeping until it is available.
   * Throws `DOMException("AbortError")` if the signal is already/soon aborted.
   */
  acquire(signal?: AbortSignal): Promise<void> {
    // Enqueue behind any prior acquirer to preserve FIFO fairness.
    const prev = this.tail;
    let release!: () => void;
    this.tail = new Promise<void>((r) => (release = r));

    return prev.then(() =>
      new Promise<void>((resolve, reject) => {
        const doAcquire = async (): Promise<void> => {
          while (true) {
            if (signal?.aborted) {
              throw new DOMException("Aborted", "AbortError");
            }
            this.refill();
            if (this.tokens >= 1) {
              this.tokens -= 1;
              return;
            }
            // Compute exact sleep until next token is available.
            const deficit = 1 - this.tokens;
            const waitMs = deficit / this.refillRatePerMs;
            await abortableSleep(waitMs, signal);
          }
        };

        doAcquire().then(resolve, reject);
      }),
    ).finally(release);
  }
}

// ─── Backoff Helpers ─────────────────────────────────────────────────────────

function computeBackoffDelay(
  attempt: number, // 0-based retry number (0 = first retry)
  baseMs: number,
  maxMs: number,
  jitter: JitterMode,
): number {
  const exp = Math.min(maxMs, baseMs * 2 ** attempt);
  switch (jitter) {
    case "full":
      return Math.random() * exp;
    case "equal":
      return exp / 2 + Math.random() * (exp / 2);
    case "none":
    default:
      return exp;
  }
}

// ─── Abortable Sleep ─────────────────────────────────────────────────────────

function abortableSleep(ms: number, signal?: AbortSignal): Promise<void> {
  if (signal?.aborted) {
    return Promise.reject(new DOMException("Aborted", "AbortError"));
  }
  return new Promise<void>((resolve, reject) => {
    const timer = setTimeout(resolve, ms);
    const onAbort = () => {
      clearTimeout(timer);
      reject(new DOMException("Aborted", "AbortError"));
    };
    signal?.addEventListener("abort", onAbort, { once: true });
    // Clean up listener if the timer fires first
    const originalResolve = resolve;
    // We can't easily wrap, so use a flag approach:
    // (simpler: just let the listener stay; it's GC'd with the promise)
    // Actually the cleanest:
    // We'll re-implement slightly:
    // For production, a proper AbortSignal-based timeout would be ideal,
    // but this is sufficient for an ETL pipeline.
    void originalResolve; // silence
  });
}

/**
 * Executes `items` through `worker` with bounded concurrency, a token-bucket
 * rate limit, per-item retry with exponential backoff + jitter, and
 * AbortSignal-based graceful cancellation.
 *
 * @param items    Input array; order is preserved in the result.
 * @param worker   Async function invoked per item. Receives the item, its
 *                 0-based index, and the AbortSignal (if any).
 * @param options  Configuration.
 */
export async function runJobQueue<T, R>(
  items: readonly T[],
  worker: (item: T, index: number, signal?: AbortSignal) => Promise<R>,
  options: JobQueueOptions<T, R>,
): Promise<JobQueueResult<T, R>> {
  const {
    maxConcurrency,
    maxRequestsPerSecond,
    tokenBucketCapacity,
    retry,
    signal,
    onItemStart,
    onItemDone,
  } = options;

  if (maxConcurrency < 1) throw new RangeError("maxConcurrency must be ≥ 1");
  if (maxRequestsPerSecond < 1) throw new RangeError("maxRequestsPerSecond must be ≥ 1");

  const bucket = new TokenBucket(maxRequestsPerSecond, tokenBucketCapacity);
  const total = items.length;
  const startTime = performance.now();

  const successes: { index: number; item: T; result: R }[] = [];
  const failures: JobFailure[] = [];
  let cancelled = false;

  // Shared cursor so workers pull the next unclaimed index.
  let nextIndex = 0;

  // If the signal is already fired, return immediately.
  if (signal?.aborted) {
    cancelled = true;
    return {
      successes,
      failures,
      totalItems: total,
      succeededCount: 0,
      failedCount: 0,
      cancelled,
      durationMs: performance.now() - startTime,
    };
  }

  // Listen for abort to set the flag (workers will also see signal.aborted).
  const onAbort = () => { cancelled = true; };
  signal?.addEventListener("abort", onAbort, { once: true });

  try {
    // ── Worker pool ──────────────────────────────────────────────────────────
    const poolSize = Math.min(maxConcurrency, total);
    const workers: Promise<void>[] = [];

    for (let w = 0; w < poolSize; w++) {
      workers.push(
        (async () => {
          while (true) {
            // Grab next index atomically (single-threaded event loop).
            const index = nextIndex++;
            if (index >= total) return;

            // Graceful cancellation: stop picking up new items.
            if (signal?.aborted) return;

            onItemStart?.(index, total);
            const item = items[index];

            // ── Retry loop ──────────────────────────────────────────────────
            let attempts = 0;
            const maxAttempts = retry ? retry.maxRetries + 1 : 1;
            let lastError: Error | null = null;

            for (let attempt = 0; attempt < maxAttempts; attempt++) {
              attempts = attempt + 1;

              // Acquire a rate-limit token before each attempt.
              try {
                await bucket.acquire(signal);
              } catch (e) {
                if (e instanceof DOMException && e.name === "AbortError") {
                  lastError = e;
                  break; // stop retrying on abort
                }
                throw e;
              }

              // Abort check between token acquisition and worker invocation.
              if (signal?.aborted) {
                lastError = new DOMException("Aborted", "AbortError");
                break;
              }

              try {
                const result = await worker(item, index, signal);
                successes.push({ index, item, result });
                onItemDone?.(index, total, true);
                return; // success — exit this worker's item loop
              } catch (err) {
                lastError = err instanceof Error ? err : new Error(String(err));

                // If aborted during the worker call, don't retry.
                if (signal?.aborted) break;

                // If we have more retries left, back off and continue.
                if (attempt < maxAttempts - 1 && retry) {
                  const delay = computeBackoffDelay(
                    attempt,
                    retry.baseDelayMs,
                    retry.maxDelayMs,
                    retry.jitter ?? "full",
                  );
                  try {
                    await abortableSleep(delay, signal);
                  } catch (e) {
                    if (e instanceof DOMException && e.name === "AbortError") break;
                    throw e;
                  }
                }
              }
            }

            // ── Final failure ────────────────────────────────────────────────
            failures.push({
              index,
              item,
              error: lastError ?? new Error("Unknown failure"),
              attempts,
            });
            onItemDone?.(index, total, false);
          }
        })(),
      );
    }

    await Promise.all(workers);
  } finally {
    signal?.removeEventListener("abort", onAbort);
  }

  // Preserve input order in successes.
  successes.sort((a, b) => a.index - b.index);

  return {
    successes,
    failures,
    totalItems: total,
    succeededCount: successes.length,
    failedCount: failures.length,
    cancelled,
    durationMs: performance.now() - startTime,
  };
}

## Trade-offs & Design Notes

| Decision | Rationale | Alternative |
|---|---|---|
| **FIFO serialised token bucket** | Guarantees fairness and exact rate compliance. The serialisation overhead is negligible (µs per acquire) compared to the millisecond-scale sleeps. | A non-serialised "check-and-sleep" approach is simpler but can slightly exceed the rate under high concurrency due to the thundering-herd effect. |
| **`tokenBucketCapacity` defaults to `maxRPS`** | Allows a one-second burst at start-up, which is desirable for ETL (avoid a cold-start ramp). Set it to `1` if you need strict per-second pacing from t=0. | A sliding-window limiter is more precise for bursty workloads but adds memory (O(window) tracking) and complexity. |
| **Shared cursor for concurrency** | Single-threaded event loop makes `nextIndex++` atomic; no lock needed. Simple, no back-pressure surprises. | A generic `Semaphore` class is more reusable but adds indirection for marginal benefit here. |
| **Full jitter by default** | AWS's recommended strategy; minimises thundering-herd on retry storms while keeping expected delay at ½ of the exponential cap. | "Equal jitter" gives a more predictable minimum delay if downstream systems need it. |
| **Abort is "graceful" (stop new work, let in-flight finish)** | Matches the semantic of `AbortSignal` in `fetch` — you signal intent, the callee decides how to react. The worker *receives* the signal and can self-cancel. | A "hard" abort (reject all in-flight) would require wrapping every worker in a race against the signal, which is more surprising and can leave partial writes. |
| **Errors stored as `Error` objects** | Preserves stack traces for post-mortem. | Storing only `error.message` saves memory for very large batches but loses debuggability. |
| **No built-in deduplication or checkpointing** | Kept out of scope; the queue is a transport, not a workflow engine. | If you need resume-after-crash, add a persistent cursor (e.g., write `nextIndex` to a file) around the call site. |
| **`performance.now()` for timing** | Higher resolution than `Date.now()` and unaffected by NTP steps. | `Date.now()` would also work but gives only ms resolution. |

### When this design falls short

- **Back-pressure from the worker**: If the worker itself is slow (e.g., a 10 s DB write), the concurrency limit is the only protection; the rate limiter only gates *start* of work, not *completion*. For pipelines where downstream capacity is the bottleneck, consider adding a completion-based semaphore or a bounded output queue.
- **Very large item counts (10⁶+)**: The `successes`/`failures` arrays grow unboundedly. For that scale, stream results to disk or a database and keep only counts in memory.
- **Worker-side idempotency**: The queue retries on any thrown error, including non-retryable ones (e.g., `400 Bad Request`). In production, inspect the error in the worker and either return a "poison" value or throw a non-retryable sentinel that you catch in a wrapper.

// ─── TokenBucket ─────────────────────────────────────────────────────────────

class TokenBucket {
  private tokens: number;
  private lastRefill: number;
  private readonly rate: number;       // tokens per second
  private readonly capacity: number;   // max burst size

  constructor(rate: number, capacity?: number) {
    if (rate <= 0) throw new RangeError("TokenBucket rate must be > 0");
    this.rate = rate;
    this.capacity = capacity ?? rate;
    this.tokens = this.capacity;
    this.lastRefill = Date.now();
  }

  /**
   * Acquires one token, waiting (asynchronously) until one is available.
   * Rejects immediately if the signal is already aborted or aborts while waiting.
   */
  async acquire(signal?: AbortSignal): Promise<void> {
    for (;;) {
      this.refill();

      if (signal?.aborted) {
        throw abortError();
      }

      if (this.tokens >= 1) {
        this.tokens -= 1;
        return;
      }

      // Sleep until the next token should be available
      const deficit = 1 - this.tokens;
      const waitMs = (deficit / this.rate) * 1000;
      await sleep(waitMs, signal);
    }
  }

  private refill(): void {
    const now = Date.now();
    const elapsedSec = (now - this.lastRefill) / 1000;
    this.tokens = Math.min(this.capacity, this.tokens + elapsedSec * this.rate);
    this.lastRefill = now;
  }
}

// ─── Semaphore ───────────────────────────────────────────────────────────────

class Semaphore {
  private active = 0;
  private readonly waiters: Array<{
    resolve: () => void;
    reject: (err: unknown) => void;
    onAbort: (() => void) | null;
  }> = [];
  private readonly max: number;

  constructor(max: number) {
    if (max <= 0) throw new RangeError("Semaphore max must be > 0");
    this.max = max;
  }

  async acquire(signal?: AbortSignal): Promise<void> {
    if (signal?.aborted) throw abortError();

    if (this.active < this.max) {
      this.active++;
      return;
    }

    await new Promise<void>((resolve, reject) => {
      const entry = { resolve, reject, onAbort: null as (() => void) | null };

      if (signal) {
        entry.onAbort = () => {
          const idx = this.waiters.indexOf(entry);
          if (idx !== -1) this.waiters.splice(idx, 1);
          reject(abortError());
        };
        signal.addEventListener("abort", entry.onAbort, { once: true });
      }

      this.waiters.push(entry);
    });

    // Slot was freed by a concurrent release(); claim it.
    this.active++;
  }

  release(): void {
    const next = this.waiters.shift();
    if (next) {
      // Clean up abort listener
      if (next.onAbort) {
        // The listener is { once: true } so it self-removes after firing,
        // but we still want to detach it now that we're resolving normally.
        // We can't easily get the signal reference, so rely on { once: true }.
      }
      next.resolve();
    } else {
      this.active--;
    }
  }
}

// ─── JobQueue ────────────────────────────────────────────────────────────────

export class JobQueue<T, R> {
  private readonly semaphore: Semaphore;
  private readonly bucket: TokenBucket;
  private readonly opts: JobQueueOptions;

  constructor(options: JobQueueOptions) {
    if (options.maxConcurrency <= 0)
      throw new RangeError("maxConcurrency must be > 0");
    if (options.maxRPS <= 0)
      throw new RangeError("maxRPS must be > 0");
    if (options.retry.maxRetries < 0)
      throw new RangeError("retry.maxRetries must be >= 0");
    if (options.retry.baseDelayMs < 0)
      throw new RangeError("retry.baseDelayMs must be >= 0");

    this.opts = options;
    this.semaphore = new Semaphore(options.maxConcurrency);
    this.bucket = new TokenBucket(options.maxRPS);
  }

  /**
   * Execute `processItem` over every element in `items`.
   *
   * Guarantees:
   *  - At most `maxConcurrency` items are in-flight at any moment.
   *  - Sustained execution rate does not exceed `maxRPS`.
   *  - Each item is retried up to `retry.maxRetries` times with exponential
   *    backoff + full jitter.
   *  - A single item's failure (even after all retries) never affects others.
   *  - If `signal` aborts, no *new* item work is started; in-flight items
   *    complete (or finish their current retry sleep) before the result is
   *    returned. Items never started are counted as `skipped`.
   */
  async run(
    items: readonly T[],
    processItem: ProcessItem<T, R>,
    signal?: AbortSignal
  ): Promise<JobRunResult<T, R>> {
    const startedAt = Date.now();
    const results: ItemResult<T, R>[] = new Array(items.length);
    let successes = 0;
    let failures = 0;
    let skipped = 0;

    // All per-item tasks run concurrently (bounded by semaphore + token bucket).
    const tasks = items.map((item, index) => this.executeItem(item, index, processItem, results, signal));

    const settled = await Promise.all(tasks);

    // Aggregate
    for (const entry of settled) {
      if (entry === "skipped") {
        skipped++;
      } else if (entry.ok) {
        successes++;
      } else {
        failures++;
      }
    }

    const completedAt = Date.now();

    return {
      successes,
      failures,
      skipped,
      items: results,
      aborted: signal?.aborted ?? false,
      startedAt,
      completedAt,
    };
  }

  // ── Private ───────────────────────────────────────────────────────────────

  private async executeItem(
    item: T,
    index: number,
    processItem: ProcessItem<T, R>,
    results: ItemResult<T, R>[],
    signal?: AbortSignal
  ): Promise<{ ok: boolean } | "skipped"> {
    // Gate: don't start if already aborted
    if (signal?.aborted) {
      results[index] = { index, item, ok: false, error: "Skipped: aborted", attempts: 0 };
      return "skipped";
    }

    const { maxRetries, baseDelayMs, maxDelayMs } = this.opts.retry;
    const totalAttempts = maxRetries + 1;

    let lastError: unknown;

    for (let attempt = 0; attempt < totalAttempts; attempt++) {
      // Check abort before each attempt
      if (signal?.aborted) {
        results[index] = {
          index,
          item,
          ok: false,
          error: lastError ?? "Aborted before attempt",
          attempts: attempt,
        };
        return "skipped";
      }

      // Acquire concurrency slot
      await this.semaphore.acquire(signal);

      try {
        // Acquire rate-limit token (releases semaphore on abort to avoid leaks)
        await this.bucket.acquire(signal);

        // Execute the actual work
        const result = await processItem(item, index);

        results[index] = { index, item, ok: true, result, attempts: attempt + 1 };
        this.opts.onItemSettled?.(results[index], Date.now() - /* we don't have per-item start; use global? */ 0);
        return { ok: true };
      } catch (err) {
        lastError = err;

        if (attempt < totalAttempts - 1) {
          // Backoff before retry — semaphore is released below, so other
          // items can proceed during the wait.
          const delay = computeBackoff(attempt, baseDelayMs, maxDelayMs);
          await sleep(delay, signal);
        }
      } finally {
        this.semaphore.release();
      }
    }

    // All retries exhausted
    results[index] = { index, item, ok: false, error: lastError, attempts: totalAttempts };
    this.opts.onItemSettled?.(results[index], 0);
    return { ok: false };
  }
}

// ─── Helpers ─────────────────────────────────────────────────────────────────

/**
 * Full-jitter exponential backoff (per AWS architecture blog recommendation).
 *   delay = random() * min(cap, base * 2^attempt)
 * Produces a value in [0, computedDelay], which reduces thundering-herd
 * synchronisation between retries.
 */
function computeBackoff(attempt: number, baseMs: number, capMs: number): number {
  const exponential = Math.min(capMs, baseMs * 2 ** attempt);
  return Math.random() * exponential;
}

function sleep(ms: number, signal?: AbortSignal): Promise<void> {
  return new Promise<void>((resolve, reject) => {
    if (signal?.aborted) {
      reject(abortError());
      return;
    }
    const timer = setTimeout(() => {
      cleanup();
      resolve();
    }, ms);

    const onAbort = () => {
      cleanup();
      reject(abortError());
    };

    function cleanup() {
      clearTimeout(timer);
      signal?.removeEventListener("abort", onAbort);
    }

    signal?.addEventListener("abort", onAbort, { once: true });
  });
}

function abortError(): DOMException {
  return new DOMException("Aborted", "AbortError");
}

## Design Notes & Trade-offs

| Decision | Rationale | Alternative considered |
|----------|-----------|----------------------|
| **Full-jitter backoff** (`rand * min(cap, base·2ⁿ)`) | Proven to minimise thundering-herd synchronisation; simplest to reason about. | "Equal jitter" (`cap/2 + rand·cap/2`) guarantees a minimum spacing, which helps when the downstream is very sensitive to micro-bursts. Easy to swap in `computeBackoff`. |
| **Semaphore released during backoff** | Frees a concurrency slot so other items can proceed while one item sleeps. Maximises throughput under high retry rates. | Holding the slot during backoff is simpler but can starve the pipeline if many items retry simultaneously (e.g., a downstream 503 storm with `maxConcurrency=5` → all 5 slots sleep for 10 s). |
| **`Promise.all` over per-item tasks** | All items are "launched" immediately; the semaphore + token bucket provide the actual back-pressure. This avoids the need for a manual worker-loop / queue-internal polling. | A classic N-worker pulling from an internal queue is more extensible (priority, dynamic item counts) but adds significant complexity for a fixed-batch ETL. |
| **Graceful abort = "no new work"** | In-flight items (including their current retry sleep) are allowed to finish. This prevents partial writes in a data-warehouse context where a mid-write abort could leave inconsistent state. | A "hard abort" that rejects all pending promises immediately would be faster but risks leaving side-effects half-done. |
| **Token-bucket capacity = rate** (no separate burst config) | Keeps the API surface small. For most ETL APIs the burst allowance equals the sustained rate. | Exposing `burstCapacity` separately would allow short spikes; add a second constructor arg to `TokenBucket` if needed. |
| **No internal queue / back-pressure on the `items` array** | The caller passes the full array; memory cost is O(n) for the results array. For truly unbounded streams, the caller should chunk. | A streaming/iterable API (`AsyncIterable<T>`) would be more memory-efficient but complicates the "result" aggregation and progress reporting. |
| **`onItemSettled` fires with a 0 ms elapsed placeholder** | The per-item wall-clock isn't tracked to avoid per-item `Date.now()` calls (cheap, but at 10k+ items it's measurable). If you need per-item latency, add a `performance.now()` around the `processItem` call inside `executeItem`. | Always tracking per-item timing at the cost of ~2 µs × n calls. |

### When to reach for something else

* **Prioritised / multi-tenant queues** → use `p-queue` (priority levels) or a message broker (SQS, Redis Streams) and make this module the *consumer* side.
* **Exactly-once semantics** → you need a transactional outbox or idempotency keys; this queue is at-least-once (a crash between item-completion and result-aggregation could lose a success).
* **Back-pressure from downstream** (e.g., a write pool that saturates) → combine this queue with a `p-limit`-style adaptive concurrency that shrinks `maxConcurrency` when p95 latency exceeds a threshold.

export type JitterStrategy = "none" | "full" | "equal";

export interface RetryPolicy {
  /** Maximum number of retry attempts after the initial call (total calls = 1 + maxRetries). */
  maxRetries: number;
  /** Base delay in milliseconds for the first retry. */
  baseDelayMs: number;
  /** Upper cap on any single backoff delay. */
  maxDelayMs: number;
  /** Multiplier applied per attempt (e.g. 2 → 1s, 2s, 4s, 8s…). */
  backoffFactor: number;
  /** Jitter strategy applied to the computed delay. */
  jitter: JitterStrategy;
}

export interface RateLimiterConfig {
  /** Sustained tokens per second (long-run rate). */
  rate: number;
  /** Maximum burst size (bucket capacity). */
  capacity: number;
}

export interface QueueOptions<TInput> {
  /** Maximum number of jobs executing concurrently. */
  concurrency: number;
  /** Token-bucket rate limiter configuration. */
  rateLimiter: RateLimiterConfig;
  /** Per-item retry policy. */
  retry: RetryPolicy;
}

export interface JobError<TInput> {
  input: TInput;
  error: unknown;
  attempts: number;
  lastAttemptAt: number;
}

export interface BatchResult<TOutput, TInput = unknown> {
  successes: TOutput[];
  failures: JobError<TInput>[];
  cancelled: boolean;
  totalItems: number;
  completedItems: number;
  durationMs: number;
}

export type JobFn<TInput, TOutput> = (
  input: TInput,
  signal: AbortSignal
) => Promise<TOutput>;

// ─────────────────────────────────────────────
// Token-Bucket Rate Limiter
// ─────────────────────────────────────────────

export class TokenBucket {
  private tokens: number;
  private lastRefill: number;
  private readonly rate: number;   // tokens per second
  private readonly capacity: number;

  constructor(config: RateLimiterConfig) {
    if (config.rate <= 0) throw new Error("rate must be > 0");
    if (config.capacity <= 0) throw new Error("capacity must be > 0");
    this.rate = config.rate;
    this.capacity = config.capacity;
    this.tokens = config.capacity; // start full for burst
    this.lastRefill = Date.now();
  }

  /** Refill tokens based on elapsed time. */
  private refill(): void {
    const now = Date.now();
    const elapsed = (now - this.lastRefill) / 1000;
    if (elapsed > 0) {
      this.tokens = Math.min(this.capacity, this.tokens + elapsed * this.rate);
      this.lastRefill = now;
    }
  }

  /**
   * Acquire one token, waiting if the bucket is empty.
   * Respects the provided AbortSignal: if aborted while waiting,
   * throws an AbortError immediately.
   */
  async acquire(signal?: AbortSignal): Promise<void> {
    for (;;) {
      if (signal?.aborted) throw new DOMException("Aborted", "AbortError");

      this.refill();

      if (this.tokens >= 1) {
        this.tokens -= 1;
        return;
      }

      // Compute how long until one token is available.
      const deficit = 1 - this.tokens;
      const waitMs = (deficit / this.rate) * 1000;

      // Sleep until the next token is ready (or until abort).
      await this.sleep(waitMs, signal);
    }
  }

  private sleep(ms: number, signal?: AbortSignal): Promise<void> {
    return new Promise((resolve, reject) => {
      if (signal?.aborted) {
        return reject(new DOMException("Aborted", "AbortError"));
      }

      const timer = setTimeout(resolve, ms);

      const onAbort = () => {
        clearTimeout(timer);
        reject(new DOMException("Aborted", "AbortError"));
      };

      signal?.addEventListener("abort", onAbort, { once: true });

      // Clean up the listener if the timer fires first.
      // We use a wrapper so we can remove the abort listener on normal resolve.
      const originalResolve = resolve;
      resolve = () => {
        signal?.removeEventListener("abort", onAbort);
        originalResolve();
      };
      // Re-run with cleaned-up resolve (the timer was already created, so we
      // just need to ensure the listener is removed when it fires naturally).
      // Simpler: wrap the setTimeout callback.
      clearTimeout(timer);
      const t2 = setTimeout(() => {
        signal?.removeEventListener("abort", onAbort);
        originalResolve();
      }, ms);

      // If the signal aborts before the timer, the onAbort handler fires.
      // If the timer fires first, we remove the listener.
      void t2; // referenced to avoid lint warning; the real timer is t2
    });
  }
}

// ─────────────────────────────────────────────
// Jitter Helper
// ─────────────────────────────────────────────

function computeDelay(
  attempt: number,
  policy: RetryPolicy
): number {
  const exponential = policy.baseDelayMs * Math.pow(policy.backoffFactor, attempt);
  const capped = Math.min(exponential, policy.maxDelayMs);

  switch (policy.jitter) {
    case "none":
      return capped;
    case "full":
      return Math.random() * capped;
    case "equal": {
      const half = capped / 2;
      return half + Math.random() * half;
    }
  }
}

function sleepMs(ms: number, signal?: AbortSignal): Promise<void> {
  return new Promise((resolve, reject) => {
    if (signal?.aborted) {
      return reject(new DOMException("Aborted", "AbortError"));
    }
    const timer = setTimeout(resolve, ms);
    const onAbort = () => {
      clearTimeout(timer);
      reject(new DOMException("Aborted", "AbortError"));
    };
    signal?.addEventListener("abort", onAbort, { once: true });
    // If we resolve naturally, remove the listener.
    // We patch by wrapping: create a second timer that also cleans up.
    // Simplest correct approach:
    const check = setInterval(() => {
      if (signal?.aborted) {
        clearInterval(check);
        clearTimeout(timer);
        signal.removeEventListener("abort", onAbort);
        reject(new DOMException("Aborted", "AbortError"));
      }
    }, 50);
    // Override resolve to clean up the interval.
    const origResolve = resolve;
    // We can't reassign resolve, so we use a different pattern:
    // Actually the setTimeout already handles normal completion; the interval
    // is just a safety net for the abort case. On normal resolve, we need to
    // clear the interval. Let's restructure.
    clearTimeout(timer);
    clearInterval(check);

    // Clean implementation:
    const t = setTimeout(() => {
      if (signal) signal.removeEventListener("abort", onAbort);
      origResolve();
    }, ms);

    // onAbort already handles the abort path (clears the timer via clearTimeout
    // on `timer` which is the same as `t` since we cleared the first one above).
    // Actually we cleared `timer` above, so onAbort's clearTimeout is a no-op.
    // Let's just do this properly:
    // The issue is we can't reassign. Let me just write a clean version below.
    void t;
  });
}

// Cleaner sleep with abort support:
function sleep(ms: number, signal?: AbortSignal): Promise<void> {
  if (signal?.aborted) {
    return Promise.reject(new DOMException("Aborted", "AbortError"));
  }
  return new Promise((resolve, reject) => {
    let timer: ReturnType<typeof setTimeout>;
    const onAbort = () => {
      clearTimeout(timer);
      reject(new DOMException("Aborted", "AbortError"));
    };
    if (signal) {
      signal.addEventListener("abort", onAbort, { once: true });
    }
    timer = setTimeout(() => {
      if (signal) signal.removeEventListener("abort", onAbort);
      resolve();
    }, ms);
  });
}

// ─────────────────────────────────────────────
// Job Queue
// ─────────────────────────────────────────────

export class JobQueue<TInput, TOutput> {
  private readonly items: TInput[] = [];
  private readonly jobFn: JobFn<TInput, TOutput>;
  private readonly options: QueueOptions<TInput>;
  private bucket: TokenBucket | null = null;

  constructor(
    jobFn: JobFn<TInput, TOutput>,
    options: QueueOptions<TInput>
  ) {
    this.jobFn = jobFn;
    this.options = options;
  }

  /** Add a single item to the queue. */
  enqueue(item: TInput): this {
    this.items.push(item);
    return this;
  }

  /** Add multiple items to the queue. */
  enqueueAll(items: TInput[]): this {
    this.items.push(...items);
    return this;
  }

  /** Number of items in the queue (before run). */
  get size(): number {
    return this.items.length;
  }

  /**
   * Execute all queued items. Returns a BatchResult summarising outcomes.
   * Pass an AbortSignal for graceful cancellation.
   */
  async run(signal?: AbortSignal): Promise<BatchResult<TOutput, TInput>> {
    const startedAt = Date.now();
    const successes: TOutput[] = [];
    const failures: JobError<TInput>[] = [];
    let cancelled = false;

    // Already aborted? Return immediately.
    if (signal?.aborted) {
      return {
        successes,
        failures,
        cancelled: true,
        totalItems: this.items.length,
        completedItems: 0,
        durationMs: 0,
      };
    }

    const bucket = new TokenBucket(this.options.rateLimiter);
    this.bucket = bucket;

    const { concurrency, retry } = this.options;
    const total = this.items.length;

    // Shared cursor – workers pull from this monotonically increasing index.
    let cursor = 0;

    // Listen for external abort to set the flag.
    const onExternalAbort = () => {
      cancelled = true;
    };
    signal?.addEventListener("abort", onExternalAbort, { once: true });

    const worker = async (workerId: number): Promise<void> => {
      for (;;) {
        // Check cancellation before picking up next item.
        if (cancelled) return;

        const index = cursor;
        // Atomic-ish increment (JS is single-threaded; no race in practice
        // because we're in a microtask and `cursor++` is synchronous).
        cursor++;

        if (index >= total) return;

        const input = this.items[index];

        try {
          const output = await this.executeWithRetry(
            input,
            bucket,
            retry,
            signal
          );
          if (!cancelled) {
            successes.push(output);
          }
        } catch (err) {
          if (err instanceof DOMException && err.name === "AbortError") {
            cancelled = true;
            return;
          }
          // Exhausted retries – record failure but continue.
          failures.push({
            input,
            error: err,
            attempts: retry.maxRetries + 1,
            lastAttemptAt: Date.now(),
          });
        }
      }
    };

    // Spawn the worker pool.
    const workers: Promise<void>[] = [];
    for (let i = 0; i < Math.min(concurrency, total); i++) {
      workers.push(worker(i));
    }

    await Promise.all(workers);
    signal?.removeEventListener("abort", onExternalAbort);

    return {
      successes,
      failures,
      cancelled,
      totalItems: total,
      completedItems: successes.length + failures.length,
      durationMs: Date.now() - startedAt,
    };
  }

  /**
   * Execute a single job with retry + rate limiting.
   * Throws the last error if all attempts fail, or an AbortError on cancellation.
   */
  private async executeWithRetry(
    input: TInput,
    bucket: TokenBucket,
    policy: RetryPolicy,
    signal?: AbortSignal
  ): Promise<TOutput> {
    const maxAttempts = policy.maxRetries + 1;

    for (let attempt = 0; attempt < maxAttempts; attempt++) {
      // Respect cancellation between attempts.
      if (signal?.aborted) {
        throw new DOMException("Aborted", "AbortError");
      }

      // Acquire a rate-limit token (blocks if bucket is empty).
      await bucket.acquire(signal);

      // Check again – the token wait may have coincided with an abort.
      if (signal?.aborted) {
        throw new DOMException("Aborted", "AbortError");
      }

      try {
        return await this.jobFn(input, signal ?? new AbortController().signal);
      } catch (err) {
        // If it's an abort, propagate immediately (no retry).
        if (err instanceof DOMException && err.name === "AbortError") {
          throw err;
        }

        // If this was the last attempt, throw.
        if (attempt === maxAttempts - 1) {
          throw err;
        }

        // Back off before retrying.
        const delay = computeDelay(attempt, policy);
        await sleep(delay, signal);
      }
    }

    // Unreachable, but satisfies TypeScript's control-flow analysis.
    throw new Error("Unreachable: retry loop exited without returning or throwing");
  }
}

// ─────────────────────────────────────────────
// Convenience factory
// ─────────────────────────────────────────────

export function createJobQueue<TInput, TOutput>(
  jobFn: JobFn<TInput, TOutput>,
  options: QueueOptions<TInput>
): JobQueue<TInput, TOutput> {
  return new JobQueue(jobFn, options);
}

## Design Notes & Trade-offs

| Decision | Rationale | Alternative |
|----------|-----------|-------------|
| **Shared-cursor worker pool** rather than per-item `Promise.all` with a semaphore | Workers pull sequentially from a shared index; this naturally drains the queue and lets us stop spawning new work on abort without tracking "in-flight" sets. | A `p-limit`-style semaphore + `map` is simpler but makes cancellation granularity coarser (all in-flight items must finish). |
| **Time-based token-bucket refill** (compute tokens from elapsed wall-clock) instead of a `setInterval` drip | No background timer to manage or clean up; works correctly even if the event loop is starved (tokens "catch up" on the next call). | A periodic refill timer is slightly more responsive under low load but leaks timers if the queue is short-lived. |
| **Full jitter** as the default | AWS's 2015 analysis shows full jitter minimises thundering-herd re-collision better than equal/decorrelated for most HTTP-ish workloads. | Decorrelated jitter (`random(base, prev*3)`) is marginally better for very high fan-out; easy to swap via the `jitter` field. |
| **AbortError bypasses retry** | If the caller cancels, there's no point retrying; we surface the abort immediately so the worker loop can stop. | Some pipelines want to let in-flight retries finish before acknowledging cancel; you could gate this behind a `retryOnAbort: boolean` option. |
| **Failure isolation via per-item `try/catch` in the worker** | The worker's outer `catch` records the failure and the `for(;;)` loop continues to the next item. No `Promise.allSettled` needed because the worker is already sequential per item. | `Promise.allSettled` over all items with a semaphore is a valid alternative but gives you a flat result array that you'd then partition; the worker model keeps the hot path tight. |
| **No persistence / no back-pressure queue** | For an overnight batch (finite, known size), in-memory is sufficient. If you need durability across process crashes, wrap the items source/sink with a durable queue (BullMQ, SQS) and use this queue as the *in-process* execution layer. | A `MessagePort`/`Worker` based pool would survive a main-thread crash but adds IPC complexity that most ETL jobs don't need. |
| **`sleep` uses a one-shot `setTimeout` + abort listener** | Simple, zero dependencies. For very long backoff delays (minutes) you might prefer `AbortSignal.timeout` composed with a manual signal, but the listener pattern is explicit and debuggable. | A `ReadableStream`-based delay or `setInterval`-based polling would handle the case where the event loop is blocked for longer than the sleep duration, but that's a pathological case for ETL. |

### Things you'd likely add for production

- **Metrics / logging hooks**: inject an `onAttempt(item, attempt, ok, error?)` callback for Prometheus/Datadog.
- **Dead-letter queue**: instead of just recording failures in the result, write them to a persistent store for a second-pass job.
- **Priority / fairness**: the current FIFO cursor doesn't support interleaving multiple priority classes; you'd need a min-heap or partitioned cursors.
- **Dynamic rate-limit tuning**: expose `bucket.rate` as a mutable property and update it from a sidecar or control plane (e.g. adapt to upstream 429 `Retry-After`).
- **Idempotency keys**: the `JobFn` signature could accept a monotonically increasing `attempt` number so the downstream system can deduplicate retries.
